import json import math import os import urllib.request import urllib.error ARCHIVO = "2025.json" ENV_FILE = ".env" XP_BASE = 5 def leer_token(): # utf-8-sig descarta el BOM que a veces agrega el Bloc de notas ruta = os.path.join(os.path.dirname(os.path.abspath(__file__)), ENV_FILE) if os.path.exists(ruta): with open(ruta, encoding="utf-8-sig") as f: for linea in f: linea = linea.strip() if linea.startswith("DISCORD_TOKEN="): return linea.split("=", 1)[1].strip().strip('"').strip("'") return os.environ.get("DISCORD_TOKEN") def cargar(): if os.path.exists(ARCHIVO): with open(ARCHIVO, encoding="utf-8") as f: return json.load(f) return {"year": 2025, "server": "Campo Eléctrico", "users": []} def guardar(data): data["users"].sort(key=lambda u: u["xp"], reverse=True) with open(ARCHIVO, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) def nivel_desde_xp(xp): return math.floor(math.sqrt(xp / XP_BASE) + 1e-9) def perfil_discord(uid, token): req = urllib.request.Request( f"https://discord.com/api/v10/users/{uid}", headers={ "Authorization": f"Bot {token}", "User-Agent": "DiscordBot (campo-electrico-top, 1.0)", }, ) try: with urllib.request.urlopen(req, timeout=10) as r: return json.loads(r.read()) except urllib.error.HTTPError as e: print(f" Discord respondió {e.code} para {uid}. " "401 = token inválido, 404 = usuario no existe.") except Exception as e: print(" No pude consultar Discord:", e) return None def datos_visuales(uid, p): avatar_hash = p.get("avatar") if avatar_hash: avatar = f"https://cdn.discordapp.com/avatars/{uid}/{avatar_hash}.png?size=1024" else: avatar = f"https://cdn.discordapp.com/embed/avatars/{(int(uid) >> 22) % 6}.png" deco_asset = (p.get("avatar_decoration_data") or {}).get("asset") decoration = ( f"https://cdn.discordapp.com/avatar-decoration-presets/{deco_asset}.png?size=96" if deco_asset else None ) banner_hash = p.get("banner") banner = f"https://cdn.discordapp.com/banners/{uid}/{banner_hash}.png?size=512" if banner_hash else None accent = p.get("accent_color") color = f"#{accent:06x}" if accent else None nombre = p.get("global_name") or p.get("username") return {"name": nombre, "avatar": avatar, "decoration": decoration, "banner": banner, "color": color} def main(): data = cargar() token = leer_token() if token: print("Token de Discord cargado.\n") else: print("ATENCIÓN: no encontré DISCORD_TOKEN en .env. Sin eso no hay foto ni nombre.\n") print("Carga del top 2025. Dejá el ID vacío y dale Enter para terminar.\n") while True: uid = input("ID de Discord: ").strip() if not uid: break if not uid.isdigit(): print("El ID tiene que ser solo números.\n") continue xp_txt = input("XP que tenía: ").strip() if not xp_txt.isdigit(): print("La XP tiene que ser un número.\n") continue xp = int(xp_txt) sugerido = nivel_desde_xp(xp) nivel_txt = input(f"Nivel (Enter para {sugerido}): ").strip() nivel = int(nivel_txt) if nivel_txt.isdigit() else sugerido nombre_manual = input("Nombre (Enter para usar el de Discord): ").strip() p = perfil_discord(uid, token) if token else None if p: visuales = datos_visuales(uid, p) print(" Perfil de Discord encontrado.") else: visuales = {"name": None, "avatar": None, "decoration": None, "banner": None, "color": None} print(" Sin perfil: se guarda con el avatar por defecto.") entrada = { "user_id": uid, "name": nombre_manual or visuales["name"] or "???", "xp": xp, "level": nivel, "avatar": visuales["avatar"], "decoration": visuales["decoration"], "banner": visuales["banner"], "color": visuales["color"], "roles": [], } data["users"] = [u for u in data["users"] if u["user_id"] != uid] data["users"].append(entrada) guardar(data) print(f"Guardado: {entrada['name']} · nivel {nivel} · {xp} XP\n") print(f"Listo. Subí {ARCHIVO} a la raíz del repo cblevels.") if __name__ == "__main__": main()
